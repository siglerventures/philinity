function testOneSheet() { var url = 'https://philinity-893d2-default-rtdb.firebaseio.com/philinity/house_data/apps-script-test.json'; var response = UrlFetchApp.fetch(url, { method: 'PUT', contentType: 'application/json', payload: JSON.stringify({test: true, time: new Date().toISOString()}), muteHttpExceptions: true }); Logger.log('HTTP: ' + response.getResponseCode()); Logger.log('Body: ' + response.getContentText()); SpreadsheetApp.getUi().alert('HTTP ' + response.getResponseCode() + '\n' + response.getContentText()); } function exportToFirebase() { var FIREBASE_URL = 'https://philinity-893d2-default-rtdb.firebaseio.com'; var FIREBASE_PATH = '/philinity/house_data'; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheets = ss.getSheets(); var exported = 0, skipped = 0, errors = []; Logger.log('Starting export of ' + sheets.length + ' sheets...'); for (var s = 0; s < sheets.length; s++) { var sheet = sheets[s]; var sheetName = sheet.getName(); var key = sheetName.toLowerCase().replace(/[^a-z0-9\s-]/g,'').replace(/\s+/g,'-').replace(/-+/g,'-').replace(/^-+|-+$/g,'').substring(0,60); if (!key) { skipped++; continue; } var data = sheet.getDataRange().getValues(); if (data.length < 2) { skipped++; continue; } var headers = data[0].map(function(h, idx) { var clean = String(h).trim().toLowerCase().replace(/[^a-z0-9\s]/g,'').replace(/\s+/g,'_').substring(0,40); return clean || ('col' + idx); }); var rows = []; for (var r = 1; r < data.length; r++) { var row = data[r]; var isEmpty = row.every(function(c) { return c === '' || c === null || c === undefined; }); if (isEmpty) continue; var rowObj = {}; for (var c = 0; c < headers.length; c++) { var val = row[c]; if (val instanceof Date) val = Utilities.formatDate(val, 'America/Denver', 'MM/dd/yyyy'); var str = String(val).trim(); if (str.length > 300) str = str.substring(0, 300); if (str && str !== 'undefined' && str !== 'null' && str !== '0') rowObj[headers[c]] = str; } if (Object.keys(rowObj).length > 0) rows.push(rowObj); } if (rows.length === 0) { skipped++; continue; } var payload = JSON.stringify({ sheet_name: sheetName, row_count: rows.length, rows: rows }); var url = FIREBASE_URL + FIREBASE_PATH + '/' + key + '.json'; try { var response = UrlFetchApp.fetch(url, { method: 'PUT', contentType: 'application/json', payload: payload, muteHttpExceptions: true }); var code = response.getResponseCode(); if (code === 200) { exported++; Logger.log('OK [' + (s+1) + '/' + sheets.length + '] ' + sheetName + ' (' + rows.length + ' rows)'); } else { Logger.log('FAIL [' + sheetName + '] HTTP ' + code + ': ' + response.getContentText().substring(0,100)); errors.push(sheetName + '(HTTP ' + code + ')'); skipped++; } } catch(e) { Logger.log('ERROR [' + sheetName + ']: ' + e.message); errors.push(sheetName); skipped++; } if (s % 5 === 4) Utilities.sleep(300); } var msg = 'Done! ' + exported + ' exported, ' + skipped + ' skipped.' + (errors.length ? '\nErrors: ' + errors.join(', ') : ''); Logger.log(msg); SpreadsheetApp.getUi().alert(msg); }
